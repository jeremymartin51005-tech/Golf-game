cmake_minimum_required(VERSION 3.16)
project(GolfCore CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
file(GLOB CORE_SRC ../Source/GolfCore/Private/*.cpp)
add_library(golfcore STATIC ${CORE_SRC})
target_include_directories(golfcore PUBLIC ../Source/GolfCore/Public)
target_compile_options(golfcore PRIVATE -Wall -Wextra)
file(GLOB TEST_SRC tests/*.cpp)
add_executable(golf_tests ${TEST_SRC})
target_link_libraries(golf_tests golfcore)
file(GLOB TOOL_SRC tools/*.cpp)
if(TOOL_SRC)
  add_executable(golfcli ${TOOL_SRC})
  target_link_libraries(golfcli golfcore)
endif()
enable_testing()
add_test(NAME golf_tests COMMAND golf_tests)
